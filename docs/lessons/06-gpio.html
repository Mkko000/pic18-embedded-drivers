الدرايفر ده هو الطبقة الوحيدة في المشروع اللي بتلمس TRIS و LAT و PORT. كل حاجة تانية بتعدّي من هنا. ولو فهمته، كل درايفر جاي (ADC، Timer، UART) هيبقى نفس الشكل بالظبط.

الأنواع: الكومبايلر يساعدك

typedef enum { GPIO_LOW = 0, GPIO_HIGH = 1 } logic_t;
typedef enum { GPIO_DIRECTION_OUTPUT = 0, GPIO_DIRECTION_INPUT = 1 } direction_t;
typedef enum { GPIO_PIN0 = 0, /* ... */ GPIO_PIN7 } pin_index_t;
typedef enum { PORTA_INDEX = 0, PORTB_INDEX, PORTC_INDEX, PORTD_INDEX, PORTE_INDEX } port_index_t;

ممكن كنا نكتب الأرقام على طول. بس gpio_pin_write(2, 0, 1) مين يفهمها بعد شهر؟ gpio_pin_write(PORTC_INDEX, GPIO_PIN0, GPIO_HIGH) بتتقري زي الجملة. وبص على الـ direction: OUTPUT = 0 و INPUT = 1، نفس قيم البت في TRIS بالظبط. يعني القيمة بتتكتب في الريجستر من غير ترجمة.

جدول بدل switch

static volatile uint8 * const tris_registers[PORT_MAX_NUMBER] = {&TRISA, &TRISB, &TRISC, &TRISD, &TRISE};
static volatile uint8 * const lat_registers[PORT_MAX_NUMBER]  = {&LATA,  &LATB,  &LATC,  &LATD,  &LATE};
static volatile uint8 * const port_registers[PORT_MAX_NUMBER] = {&PORTA, &PORTB, &PORTC, &PORTD, &PORTE};

دي أهم حتة في الملف. بدل ما كل دالة فيها switch بخمس حالات، عندنا جدول عناوين. lat_registers[PORTC_INDEX] هو عنوان LATC. فـ SET_BIT(*(lat_registers[port]), pin) سطر واحد بيشتغل مع أي بورت.

اقرا التعريف من اليمين للشمال: tris_registers مصفوفة من مؤشرات const (المؤشر نفسه ثابت، محدش يغيّر الجدول)، كل مؤشر بيشاور على volatile uint8.

خلي بالككلمة volatile هنا مش اختيارية. الكومبايلر بيحب يختصر: لو شاف إنك قريت PORTB مرتين ورا بعض ومحدش كتب فيه، هيقراه مرة واحدة ويستخدم القيمة تاني. بس PORTB بيتغير لوحده لما حد يدوس الزرار. volatile بتقوله "دي حاجة بتتغير من وراك، اقراها كل مرة". من غيرها هتلاقي لوب مستني زرار، شغال مظبوط في الـ debug ومبيخلصش في الـ release.

قبل أي حاجة: الفحص

#define IS_VALID_PORT_PIN(port, pin) (IS_VALID_PORT(port) && IS_VALID_PIN(pin) && \
        ((((uint8)(port)) != PORTE_INDEX) || (((uint8)(pin)) < PORTE_PIN_MAX_NUMBER)))

كل دالة بتبدأ بالفحص ده. البورت أقل من 5؟ الرجل أقل من 8؟ ولو PORTE، الرجل أقل من 3؟ لو لأ، بترجع E_NOT_OK من غير ما تلمس أي ريجستر. ليه ده مهم؟ لأن lat_registers[7] مش موجودة. لو حد بعت 7، المؤشر هيقرا عنوان عشوائي من الذاكرة، ويكتب فيه. هتلاقي متغير في حتة تانية خالص في البرنامج اتغيّر لوحده. ده من أسوأ الـ bugs اللي ممكن تدوّر عليها.

Direct API: كل حاجة في الباراميترز

Std_ReturnType gpio_pin_setup(port_index_t port, pin_index_t pin,
                              direction_t direction, logic_t logic);
Std_ReturnType gpio_pin_write(port_index_t port, pin_index_t pin, logic_t logic);
Std_ReturnType gpio_pin_read(port_index_t port, pin_index_t pin, logic_t *logic);
Std_ReturnType gpio_pin_read_latch(port_index_t port, pin_index_t pin, logic_t *logic);
Std_ReturnType gpio_pin_toggle(port_index_t port, pin_index_t pin);

Std_ReturnType gpio_port_setup(port_index_t port, uint8 direction_mask, uint8 initial_value);
Std_ReturnType gpio_port_write(port_index_t port, uint8 logic_mask);

تنادي وتمشي. مفيش struct تحضّره. وده معناه: مفيش RAM ضايعة، وأقل تعليمات.

السر في ترتيب gpio_pin_setup

/* Step 1: Initial Logic first (glitch-free output) */
if (GPIO_DIRECTION_OUTPUT == direction) {
    if (GPIO_HIGH == logic) { SET_BIT(*(lat_registers[port]), pin); }
    else                    { CLEAR_BIT(*(lat_registers[port]), pin); }
    CLEAR_BIT(*(tris_registers[port]), pin);    /* 0 = Output */
}

الأول بيكتب في LAT، وبعدين يخلّي الرجل Output. فاكر الأداة بتاعة الدرس الأول؟ ده بالظبط: القيمة تستنى في الـ Latch، والرجل تطلع وهي جاهزة. لو عكست الترتيب، الرجل هتبقى Output للحظة بالقيمة القديمة اللي في LAT (اللي ممكن تكون أي حاجة بعد الريست)، وبعدين تتصلح. اللحظة دي اسمها glitch. مع ليد مش هتلاحظها. مع ريلاي قافل باب، أو موتور، أو Active-Low LED، هتلاحظها.

Struct API: الرجل كحاجة واحدة

typedef struct {
    uint8 port      : 3;
    uint8 pin       : 3;
    uint8 direction : 1;
    uint8 logic     : 1;
} pin_config_t;          /* 3 + 3 + 1 + 1 = 8 بت = بايت واحد */

Std_ReturnType gpio_pin_initialize(const pin_config_t *_pin_config);
Std_ReturnType gpio_pin_write_logic(const pin_config_t *_pin_config, logic_t logic);
Std_ReturnType gpio_pin_toggle_logic(const pin_config_t *_pin_config);

الأرقام اللي بعد النقطتين اسمها bitfields. بتقول للكومبايلر "الحقل ده محتاج 3 بت بس". 3 بت بتشيل من 0 لـ 7، كفاية للبورتات (0–4) وللرجول (0–7). فالـ struct كله بايت. لو كان كل حقل uint8 عادي كان هياخد 4 بايت.

الفايدة الحقيقية إنك بتعرّف الرجل مرة واحدة فوق في الكود، وبعدين تبعتها في كل حتة كمؤشر واحد بدل أربع باراميترز. ولو الرجل اتنقلت، بتغيّر سطر واحد.

const pin_config_t buzzer = { .port = PORTD_INDEX, .pin = GPIO_PIN7,
                              .direction = GPIO_DIRECTION_OUTPUT, .logic = GPIO_LOW };
gpio_pin_initialize(&buzzer);
gpio_pin_write_logic(&buzzer, GPIO_HIGH);

حتة صغيرة حصلت معانا فعلًا: الـ buzzer هنا const، يعني قاعد في الـ Flash، والليد درايفر بيبعت struct قاعد في الـ RAM لنفس الدالة. XC8 v3 لما بيلاقي مؤشر ممكن يشاور على المكانين، بيرفض يقرا الـ bitfields منه وبيطلع error (712) "can't generate code". عشان كده كل دوال الـ Struct API أول حاجة بتعملها إنها تاخد نسخة من الـ struct في الـ RAM (gpio_cfg_copy) وتشتغل عليها. بايت واحد زيادة على الـ stack، والكود بقى يتبني في كل الحالات.

Analog ولا Digital

رجول زي RA0 و RB0 ممكن تبقى دخل للـ ADC. بعد الريست أغلبها بتبقى Analog، والرجل الـ Analog لو قريتها كـ Digital بتطلع صفر دايمًا. ده من أشهر مقالب الـ PIC: الزرار على RA0 مبيستجبش، والكود سليم.

علشان كده فيه gpio_init() اللي بتحط ADCON1 PCFG = 0x0F وتخلي الكل Digital، وأول سطر في main بيناديها. وكمان gpio_pin_setup نفسها بتحوّل الرجل Digital لوحدها، إلا لو الـ ADC طالبها. ولو عايز رجل Analog بإيدك فيه gpio_pin_set_analog().

جوّه الملف فيه جدول بيقول كل رجل تبع أنهي قناة: RA0 هي AN0، و RB0 هي AN12، وهكذا. والـ PIC ده مش بيخليك تختار كل قناة لوحدها، بيخليك تقول "من AN0 لحد ANn يبقوا Analog". فالدرايفر بيدوّر على أعلى قناة اتطلبت ويحسب PCFG على أساسها. يعني لو طلبت AN3 Analog، AN0 لـ AN2 هيبقوا Analog معاها غصب عنك. ده قيد في الهاردوير، مش في الدرايفر.

Pull-ups على PORTB

الزرار محتاج مقاومة تشد الرجل لـ 5V وهو مش مضغوط، وإلا الرجل هتبقى سايبة وتقرا أي حاجة. PORTB فيه مقاومات جوّه الشريحة بتعمل كده، وبتتفتح كلها مع بعض بـ gpio_portb_pullup_enable(). فالزرار بيتوصل بين الرجل والأرضي على طول. مضغوط = 0، سايب = 1. مشروع العدّاد بيستخدم ده.

عايز تشوف الدرايفر ده بيتكتب من الصفر، نسخة ورا نسخة؟ ده الجزء التالت: الـ Direct API نسخة ورا نسخة، و الـ Struct API والمفاتيح، و Analog ولا Digital.

ليه gpio_pin_setup بتكتب في LAT قبل TRIS؟

لو TRIS الأول، الرجل هتطلع للحظة بالقيمة القديمة اللي في الـ Latch، وده glitch ممكن يحرك ريلاي أو موتور.

ندهت gpio_pin_write(PORTE_INDEX, GPIO_PIN5, GPIO_HIGH). إيه اللي هيحصل؟

PORTE فيه 3 رجول بس (RE0–RE2). الفحص IS_VALID_PORT_PIN بيمسكها قبل ما توصل للهاردوير.
