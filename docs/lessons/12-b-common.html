عندنا كود بيومض ليد بالريجسترات. لو كمّلنا كده، كل مشروع هيبقى ملف واحد طويل مليان TRISC و LATD. الدرس ده بنرتب البيت قبل ما نجيب العفش: الفولدرات، والأنواع، والماكروز.

الخطوة 1: الفولدرات

الهدف: هيكل يقدر ياخد 20 درايفر من غير ما يبقى كركبة.
pic18-embedded-drivers/
├── application.c
├── Config/        device_config.h, config_bits.c
├── Common/        mcal_std_types.h, std_macros.h, compiler.h, std_libraries.h
├── MCAL_Layer/    GPIO/  (وبعدين ADC/ و Timer/ و UART/ ...)
└── ECU_Layer/     LED/   (وبعدين Button/ و LCD/ ...)

طب ليه مش فولدر واحد اسمه drivers وخلاص؟ لأن فيه نوعين من الدرايفرات مختلفين جدًا. نوع بيكلم حاجة جوّه الشريحة (GPIO، Timer، ADC): ده بيلمس ريجسترات، ومربوط بالشريحة دي بالذات. ونوع بيكلم حاجة برّه الشريحة (ليد، زرار، شاشة): ده مالوش دعوة بالريجسترات، وبيعرف إن الليد ليها Active Low ولا High. الأول اسمه MCAL (Microcontroller Abstraction Layer)، والتاني ECU Abstraction.

مثال حقيقي: قررت تنقل المشروع من PIC18F4620 لـ PIC18F45K22 (أحدث وأرخص). في الـ K22 الـ Analog بيتحكم فيه بريجستر اسمه ANSELx لكل بورت، مش ADCON1. لو الكود كله ملف واحد، هتدوّر على كل حتة. مع الطبقات، بتعدّل في MCAL_Layer/GPIO بس، والليد والأبلكيشن مش هيحسوا.

والـ Config والـ Common؟ دول مش طبقات، دول حاجات الكل محتاجها. الـ Config فيه قرارات المشروع ده (سرعة، فيوزات، إيه مفتوح). الـ Common فيه أدوات عامة مالهاش علاقة بأي مشروع (أنواع، ماكروز). لو بدأت مشروع جديد خالص، هتاخد Common زي ما هو، وتكتب Config جديد.

الخطوة 2: الأنواع

الهدف: أنواع حجمها معروف، ونوع واحد لنتيجة كل الدوال.
#ifndef MCAL_STD_TYPES_H
#define MCAL_STD_TYPES_H

#include "std_libraries.h"
#include "compiler.h"

typedef unsigned char   uint8;
typedef signed char     sint8;
typedef unsigned short  uint16;
typedef signed short    sint16;
typedef unsigned long   uint32;
typedef signed long     sint32;

typedef uint8           Std_ReturnType;
#define E_OK            (Std_ReturnType)0x01
#define E_NOT_OK        (Std_ReturnType)0x00

#endif

ليه مش stdint.h و uint8_t؟ سؤال في محله، والـ XC8 بيدعمها. الأسامي دي جاية من AUTOSAR، وكتير من الشغل في الصناعة ماشي عليها، فالكود بيبقى شبه اللي هتشوفه في الشغل. والأهم إن عندنا مكان واحد بنقرر فيه الحجم: لو نقلنا الكود لمعالج 32 بت فيه unsigned short مش 16، بنغيّر سطر واحد. لو حبيت تستخدم uint8_t مش غلط خالص، المهم متخلطش الاتنين في نفس المشروع.

ليه Std_ReturnType؟ كل دالة في الدرايفر بترجع إذا كانت نجحت ولا لأ. لو كل واحدة رجّعت حاجة شكل (bool هنا، int هناك، -1 لما تفشل)، الكود اللي بينادي هيبقى مليان حالات. نوع واحد وقيمتين.

Include guard: الـ #ifndef ... #define ... #endif. hal_gpio.h بيعمل include لـ mcal_std_types.h، و ecu_led.h بيعمل include لـ hal_gpio.h، والأبلكيشن بيعمل include للاتنين. من غير الـ guard الملف هيتقري مرتين والـ typedef هيتكرر ويطلع خطأ.

الخطوة 3: الماكروز، نسخة ورا نسخة

الهدف: نشغّل ونطفّي بت واحد في أي ريجستر بكلمة واضحة.

النسخة الأولى اللي أي حد هيكتبها:

#define SET_BIT(REG, BIT)   REG |= 1 << BIT

شغالة في SET_BIT(LATC, 3). تعالى نكسرها:

SET_BIT(LATC, n + 1);          /* يبقى: LATC |= 1 << n + 1   ← تمام بالصدفة، الجمع أقوى من الشيفت */
if (x) SET_BIT(LATC, 0); else ...;   /* تمام */
y = SET_BIT(LATC, 0) | 0x10;   /* يبقى: y = LATC |= 1 << 0 | 0x10  ← الـ OR اتطبق على الماسك، ولّعت بت 4 كمان */

المشكلة إن الماكرو بيلزق الكلام زي ما هو، والأولوية بين العمليات بتلعب. الحل: أقواس حوالين كل باراميتر، وحوالين الماكرو كله.

-#define SET_BIT(REG, BIT)   REG |= 1 << BIT
+#define BIT_MASK                    ((uint8)1)
+#define SET_BIT(REG, BIT_POSN)      ((REG) |=  ((BIT_MASK) << (BIT_POSN)))
+#define CLEAR_BIT(REG, BIT_POSN)    ((REG) &= ~((BIT_MASK) << (BIT_POSN)))
+#define TOGGLE_BIT(REG, BIT_POSN)   ((REG) ^=  ((BIT_MASK) << (BIT_POSN)))
+#define READ_BIT(REG, BIT_POSN)     (((REG) >> (BIT_POSN)) & (BIT_MASK))

ليه (uint8)1 مش 1 وخلاص؟ الـ 1 لوحده int، وعلى الـ XC8 ده 16 بت. أدوات الفحص زي MISRA هتشتكي من خلط الأحجام. والأهم إن الاسم BIT_MASK بيقول المعنى.

والـ CLEAR بـ ~: ~(1 << 3) بيدّي 1111 0111. الـ AND معاه بيسيب كل البتات زي ما هي ويصفّر بت 3 بس.

الغلطة اللي مفيش ماكرو يحميك منها

SET_BIT(arr[i++], 0) بتتحول لـ ((arr[i++]) |= ...)، تمام هنا، مرة واحدة. بس READ_BIT لو اتكتب بطريقة فيها REG مرتين كانت هتزوّد i مرتين. القاعدة: متدّيش ماكرو أبدًا حاجة فيها ++ أو نداء دالة. التحذير مكتوب أول الملف في الريبو.

الخطوة 4: compiler.h

ملف صغير بيكشف الكومبايلر. فايدته الحقيقية بتظهر لما تيجي تختبر دالة حسابية (زي تحويل قراية ADC لدرجة حرارة) على الكمبيوتر بـ GCC قبل ما تحرقها: نفس الكود يتبني في المكانين.

#if defined(__XC8__)
    #define COMPILER_XC8
#elif defined(__GNUC__)
    #define COMPILER_GCC
#endif

#if defined(COMPILER_XC8) || defined(COMPILER_GCC)
    #define INLINE              static inline
#else
    #define INLINE              static
#endif

ليه الـ LED driver في ECU مش في MCAL؟

لو غيّرت الشريحة، الليد هي هي. اللي بيتغير هو الـ MCAL.

CLEAR_BIT(LATC, 5) و LATC كان 0xFF. النتيجة؟

~(1<<5) = 1101 1111 = 0xDF. الـ AND معاه بيصفّر بت 5 بس.

تمرين: ماكرو لأكتر من بت

اكتب ماكرو WRITE_FIELD(REG, MASK, SHIFT, VAL) يكتب قيمة في مجموعة بتات من غير ما يلمس الباقي. مثال: WRITE_FIELD(ADCON1, 0x0F, 0, 0x0E) يكتب 0x0E في البتات الأربعة اللي تحت.

اعرض الحل 
#define WRITE_FIELD(REG, MASK, SHIFT, VAL) \
    ((REG) = (uint8)(((REG) & (uint8)~(MASK)) | ((uint8)((VAL) << (SHIFT)) & (MASK))))

امسح المكان الأول بالـ AND مع عكس الماسك، وبعدين حط القيمة بعد ما تقصّها بالماسك علشان لو حد بعت قيمة كبيرة متطلعش برّه. ده بالظبط اللي الـ GPIO بيعمله مع ADCON1. لاحظ إن REG بيظهر مرتين، فالقاعدة بتاعة الـ ++ مهمة هنا أكتر.
