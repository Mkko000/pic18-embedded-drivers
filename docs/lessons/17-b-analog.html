ده أصعب جزء في الدرايفر، وأغلب الناس بتعدّيه بسطر واحد: ADCON1 = 0x0F;. السطر ده كفاية لحد يوم ما تحتاج حساس على RA0 وزرار على RB0 في نفس المشروع. الدرس ده عن اليوم ده.

المشكلة بالأرقام

من درس الداتاشيت: الـ PCFG أربع بتات، والقيمة بتقول "من AN0 لحد ANn يبقوا Analog". مفيش اختيار لكل قناة لوحدها.

PCFGAnalog
0x0Fولا واحدة
0x0EAN0
0x0DAN0..AN1
……
0x0E − nAN0..ANn
0x02 (وأقل)AN0..AN12 كلهم

يعني لو عايز RB0 (اللي هي AN12) Analog، كل القنوات من AN0 لـ AN12 هيبقوا Analog. RA0 و RA1 و RB1 وكله. أي زرار عليهم هيتقري 0.

v1: gpio_init وبس

Std_ReturnType gpio_init(void) {
    ADCON1 = 0x0F;
    return E_OK;
}

مشكلتين. الأولى إن بتات 4 و 5 في ADCON1 (VCFG) بتحدد الـ reference voltage للـ ADC، والسطر ده صفّرهم. لو درايفر الـ ADC كان ظبطهم قبلها، اتمسحوا. الحل: نكتب في الـ PCFG بس.

-    ADCON1 = 0x0F;
+    ADCON1 = (uint8)((ADCON1 & (uint8)~GPIO_PCFG_MASK) | GPIO_PCFG_ALL_DIGITAL);

التانية إن حد ممكن ينسى ينادي gpio_init، أو يناديها والـ ADC بعدها يرجّع رجول Analog. فالـ setup لازم تتأكد بنفسها.

v2: الرجل تبقى Digital لوحدها

الهدف: gpio_pin_setup على رجل Analog تحوّلها Digital، من غير ما تبوّظ قنوات الـ ADC المطلوبة.

الفكرة: الدرايفر يفتكر أنهي قنوات حد طلبها Analog بقصد. متغير 16 بت، كل بت قناة:

static uint16 analog_request_mask = 0;     /* bit n = ANn مطلوبة Analog */

ولما حاجة تتغير، نحسب PCFG من أعلى قناة مطلوبة:

static void gpio_analog_apply(void) {
    uint8 pcfg    = GPIO_PCFG_ALL_DIGITAL;
    sint8 channel;

    for (channel = GPIO_MAX_ANALOG_CHANNEL; channel >= 0; channel--) {
        if (analog_request_mask & (uint16)BIT_VALUE(channel)) {
            pcfg = (uint8)(0x0E - (uint8)channel);
            break;
        }
    }
    ADCON1 = (uint8)((ADCON1 & (uint8)~GPIO_PCFG_MASK) | pcfg);
}

ليه sint8 للعداد؟ لأن اللوب نازل لحد 0، وبعد 0 لازم يبقى −1 علشان يقف. لو كان uint8، بعد 0 هيبقى 255، والشرط >= 0 دايمًا صح، واللوب مش هيخلص. نفس غلطة الـ 256 في مشروع العداد بالمقلوب.

وجدول بيربط كل رجل بقناتها (شفناه في درس الداتاشيت). NA للرجول اللي مالهاش.

الحالة اللي متتحلش

إنت طلبت AN4 (RA5) Analog لحساس، وبعدين ندهت gpio_pin_set_digital على RA0 (AN0). الدرايفر يشيل AN0 من المطلوب، ويحسب، فيلاقي AN4 لسه مطلوبة، يعني PCFG هيفضل يخلي AN0..AN4 Analog. RA0 هتفضل Analog غصب عن الكل.

الدرايفر مش هيكدب عليك. بيرجّع E_NOT_OK:

static Std_ReturnType gpio_pin_make_digital(port_index_t port, pin_index_t pin) {
    Std_ReturnType ret     = E_OK;
    uint8          channel = analog_channel_map[port][pin];

    if (GPIO_NO_ANALOG_CHANNEL != channel) {
        analog_request_mask &= (uint16)~BIT_VALUE(channel);
        gpio_analog_apply();

        /* Threshold limitation: still analog if a higher channel is analog */
        if ((sint8)channel <= gpio_analog_highest_hw_channel()) {
            ret = E_NOT_OK;
        }
    }
    return ret;
}

وده درس في التصميم أهم من الكود: لما الهاردوير مش بيسمح بحاجة، الدرايفر يقول. والحل الحقيقي في التوصيل: الحساسات على القنوات الصغيرة (AN0، AN1)، والزراير على الرجول اللي مالهاش Analog أصلًا (PORTC، PORTD، RB5..RB7).

gpio_pin_auto_digital: من غير ما يزعّل الـ ADC

static inline void gpio_pin_auto_digital(port_index_t port, pin_index_t pin) {
#if GPIO_ANALOG_CONFIG_ENABLE == CONFIG_ENABLE
    uint8 channel = analog_channel_map[port][pin];
    if (GPIO_NO_ANALOG_CHANNEL != channel) {
    #if ADC_MODULE_ENABLE == CONFIG_ENABLE
        if (!(analog_request_mask & (uint16)BIT_VALUE(channel)))
    #endif
        {
            (void)gpio_pin_make_digital(port, pin);
        }
    }
#else
    UNUSED(port);
    UNUSED(pin);
#endif
}

اقراها كده: لو الرجل ليها قناة، والـ ADC مفتوح والقناة دي مطلوبة، سيبها. غير كده خليها Digital. الـ (void) قدام النداء معناها "شايف النتيجة ومش محتاجها هنا"، لأن setup هتكمّل في الحالتين. والـ UNUSED في حالة إن الميزة مقفولة علشان الكومبايلر ميشتكيش من باراميترز مش مستخدمة.

غلطات مشهورة

كتبت ADCON1 = 0x0F في الأبلكيشن بعد ما الـ ADC اتظبط: الـ VCFG اتمسح والقراية بقت غلط بفرق صغير ومحيّر.

زرار على RB0 والـ PBADEN = ON والـ gpio_init مش متنادية: الزرار دايمًا 0.

حساس على AN12 (RB0): كده كل AN0..AN12 بقوا Analog. حط الحساس على AN0.

طلبت AN2 Analog. إيه قيمة PCFG؟

0x0E − 2 = 0x0C، يعني AN0..AN2 Analog.

ليه عداد اللوب في gpio_analog_apply من نوع sint8؟

لوب نازل بشرط >= 0 مع unsigned لوب لا نهائي.

تمرين: خطّط البورد

مشروع فيه: حساس حرارة LM35، و potentiometer، و 4 زراير، و 8 ليدات. وزّعهم على الرجول بحيث مفيش أي تعارض Analog/Digital.

اعرض حل مقترح

LM35 على RA0 (AN0)، والـ pot على RA1 (AN1). كده PCFG = 0x0D، وبس AN0 و AN1 Analog، وكل القنوات اللي فوقهم Digital. الزراير على RB4..RB7 مع الـ pull-ups الداخلية: RB4 ليها قناة (AN11) بس هي فوق AN1 فهتفضل Digital، والأربعة دول كمان ليهم interrupt-on-change هتحتاجه بعدين. الليدات على PORTD كله، وسيب PORTC للـ UART و I2C قدّام.
