أول ما تفتح الريبو هتلاقي فولدرات كتير، وممكن تقول "كل ده علشان ليد؟". عندك حق، لو المشروع هيفضل ليد. بس الريبو ده معمول علشان يكبر: زرار، شاشة LCD، ADC، UART. والطريقة دي هي اللي بتخلي الإضافة سهلة بدل ما كل درايفر جديد يكسر اللي قبله.

العمارة والأدوار

فكّر في عمارة. الـ Application هو الساكن: عايز نور في الصالة، وخلاص. الـ ECU هو الكهربائي: يعرف إن "نور الصالة" ده ليد على RC0 بتنوّر بالـ HIGH. الـ MCAL هو لوحة الكهربا نفسها: يعرف TRISC و LATC وبس، وميعرفش يعني إيه ليد. والـ Config و Common دول الكود والمواصفات اللي الكل ماشي عليها.

application.c ← "نوّر الأحمر" │ ECU_Layer/LED ← led_on(&led_red) │ MCAL_Layer/GPIO ← gpio_pin_write(PORTC, PIN0, HIGH) │ TRISC / LATC / PORTC ← الهاردوير Config/ و Common/ ← الكل بيستخدمهم

القاعدة الذهبية: كل طبقة بتكلم اللي تحتها بس. الـ application مبيلمسش ريجستر، والـ ECU مبيلمسش ريجستر. لو بكرة نقلت الكود لـ PIC تانية، هتغيّر الـ MCAL وبس، والليد والأبلكيشن زي ما هم.

اللي عايز يعرف أصل الفكرة: دي نسخة مبسطة من AUTOSAR، اللي شركات العربيات بتستخدمه. هناك الطبقات أكتر بكتير، بس الروح واحدة.

Config/device_config.h: مكان واحد لكل القرارات

ده الملف اللي بتفتحه أول حاجة. فيه سرعة الكريستالة، ونوع الأوسيليتور، وفيه مفاتيح تفتح وتقفل كل درايفر:

#define _XTAL_FREQ          4000000UL
#define OSC_MODE            OSC_MODE_HS

#define GPIO_MODULE_ENABLE              CONFIG_ENABLE
#define GPIO_STRUCT_API_ENABLE          CONFIG_ENABLE
#define GPIO_DIRECT_API_ENABLE          CONFIG_ENABLE
#define GPIO_ANALOG_CONFIG_ENABLE       CONFIG_ENABLE
#define GPIO_INIT_ALL_DIGITAL           CONFIG_ENABLE
#define GPIO_PORTB_PULLUP_ENABLE        CONFIG_ENABLE

#define ADC_MODULE_ENABLE               CONFIG_DISABLE
#define LED_MODULE_ENABLE               CONFIG_ENABLE

_XTAL_FREQ مش لينا، ده للكومبايلر. الـ __delay_ms(500) بتحسب عدد اللفات من الرقم ده. لو كتبت 4MHz والكريستالة 20MHz، كل delay هيبقى أسرع خمس مرات، والـ Morse بتاعك هيبقى صفارة.

المفاتيح بتشتغل بالـ #if. لما تقفل حاجة، الكومبايلر حرفيًا مش بيشوف الكود بتاعها، فمش بياخد ولا بايت من الـ Flash. في شريحة الـ Flash فيها 64KB، ده بيفرق لما المشروع يكبر.

الـ #error: الملف بيحمي نفسه

آخر الملف فيه فحوصات:

#if (LED_MODULE_ENABLE == CONFIG_ENABLE) && (GPIO_MODULE_ENABLE != CONFIG_ENABLE)
    #error "LED_MODULE requires GPIO_MODULE to be enabled!"
#endif

الليد مبنية فوق الـ GPIO. لو حد قفل الـ GPIO ونسي الليد شغالة، من غير السطر ده هيطلعله 40 خطأ غريب من نوع "undefined reference". مع السطر ده بيطلعله خطأ واحد بيقوله بالظبط إيه المشكلة. ده أرخص توثيق ممكن تكتبه، لأنه بيشتغل لوحده.

Config/config_bits.c: الفيوزات

الـ Configuration Bits إعدادات بتتحرق مع البرنامج ومبتتغيرش وهو شغال: نوع الأوسيليتور، الـ Watchdog، الـ Brown-out، الحماية. الـ XC8 بيقبلها في أي ملف واحد، فحطيناها في ملف لوحدها علشان متتكررش.

#if   OSC_MODE == OSC_MODE_XT
#pragma config OSC = XT         /* XT crystal (<= 4 MHz) */
#elif OSC_MODE == OSC_MODE_HS
#pragma config OSC = HS         /* HS crystal (4 - 25 MHz) */
...
#pragma config WDT   = OFF
#pragma config LVP   = OFF
#pragma config MCLRE = ON
#pragma config PBADEN = OFF

واختيار الأوسيليتور نفسه مربوط بـ OSC_MODE اللي في device_config.h، والملف بيراجع الرقم ده مع _XTAL_FREQ: لو اخترت XT مع كريستالة 8MHz، أو HSPLL والسرعة طالعة برّه 16–40MHz، هيرفض يكمل ويقولك ليه.

شوية حاجات تستاهل وقفة:

LVP = OFF: لو سبتها ON، رجل RB5 بتتحجز للبرمجة على جهد واطي، ولو اتلمست بتدخل الشريحة في وضع البرمجة. ناس كتير قعدت أيام مش فاهمة ليه البورد بتهنج لما إيدها تقرب منها.

PBADEN = OFF: رجول PORTB من RB0 لـ RB4 بتبدأ Analog بعد الريست لو دي ON. فالزرار اللي على RB0 هيتقري صفر دايمًا. الإعداد ده بيبدأها Digital.

WDT = OFF: الـ Watchdog بيعمل ريست لو الكود ماقالوش "أنا عايش" كل فترة. مفيد جدًا في جهاز حقيقي، ومزعج جدًا وإنت بتتعلم، لأن الكود هيعمل ريست كل شوية من غير سبب واضح.

MCLRE = ON: رجل MCLR هي زرار الريست. لازم عليها مقاومة 10k لـ VDD، وإلا الشريحة هتفضل في ريست.

Common/: اللغة المشتركة

mcal_std_types.h

typedef unsigned char   uint8;
typedef signed char     sint8;
typedef unsigned short  uint16;
typedef unsigned long   uint32;
typedef uint8           Std_ReturnType;

#define E_OK            (Std_ReturnType)0x01
#define E_NOT_OK        (Std_ReturnType)0x00

ليه نعمل uint8 بدل int؟ لأن int مالوش حجم ثابت في C. على الـ PIC18 بالـ XC8 هو 16 بت، وعلى الكمبيوتر 32. لما تكتب uint8 إنت عارف بالظبط إنه بايت، وفي ميكرو RAM بتاعته 4KB ده مهم. والـ PIC18 معالج 8 بت، فأي حاجة أكبر من بايت بتاخد تعليمات أكتر.

خلي بالكفي المشروع ده E_OK = 1 و E_NOT_OK = 0. ده عكس AUTOSAR الأصلي وعكس عادة لينكس (صفر يعني نجاح). علشان كده متكتبش if (!ret) وإنت فاكرها "لو مفيش خطأ". اكتب دايمًا if (E_OK == ret) بالاسم، وكده الكود صح أيًا كانت القيمة.

std_macros.h

#define BIT_MASK                    ((uint8)1)
#define SET_BIT(REG, BIT_POSN)      ((REG) |=  ((BIT_MASK) << (BIT_POSN)))
#define CLEAR_BIT(REG, BIT_POSN)    ((REG) &= ~((BIT_MASK) << (BIT_POSN)))
#define TOGGLE_BIT(REG, BIT_POSN)   ((REG) ^=  ((BIT_MASK) << (BIT_POSN)))
#define READ_BIT(REG, BIT_POSN)     (((REG) >> (BIT_POSN)) & (BIT_MASK))

خد SET_BIT(LATC, 3) كمثال. 1 << 3 بيعمل 0000 1000. والـ |= بتعمل OR: أي بت فيه 1 في الماسك هيبقى 1، والباقي زي ما هو. يعني بنولّع بت 3 من غير ما نلمس الباقي. الـ CLEAR بتقلب الماسك (1111 0111) وتعمل AND، فبت 3 بس هو اللي يتصفّر. والـ XOR في التوجل بيقلب البت اللي في الماسك بس.

شوف الأقواس حوالين كل حاجة في الماكرو. دي مش زينة. لو حد كتب SET_BIT(LATC, n+1) ومن غير أقواس، الماكرو هيبقى 1 << n+1، والأولوية هتلخبط الحساب في حالات تانية. وتعليق الملف نفسه بيحذرك متديش الماكرو حاجة فيها ++، لأن الماكرو بينسخ الكلام، فالـ x++ ممكن تتنفذ مرتين.

compiler.h

بيكشف إنت بتعمل كومبايل بإيه (XC8 ولا GCC)، ويدّيك أسماء ثابتة زي INLINE و INTERRUPT_HANDLER و NOP(). الفايدة إنك تقدر تعمل كومبايل لجزء من الكود على الكمبيوتر بـ GCC وتعمله تيست من غير بورد خالص.

عملت ADC_MODULE_ENABLE CONFIG_ENABLE بس لسه مفيش درايفر ADC. إيه اللي هيحصل للـ GPIO؟

في hal_gpio.c فيه #if ADC_MODULE_ENABLE جوّه الجزء بتاع التحويل للـ Digital. لما الـ ADC يبقى ON، الـ GPIO بيحترم الرجول اللي اتطلبت Analog. هنشوفه في الدرس الجاي.
