الليد أبسط حاجة ممكن تتوصل بميكرو. ومع ذلك ليها درايفر لوحدها. السبب إن "نوّر الليد" مش دايمًا معناها "اكتب 1". والسؤال اللي بدأ منه الدرس ده كان سؤال حقيقي وقت كتابة الكود: ليه led_initialize كانت بتستخدم struct ومعانا Direct API؟

النسخة الأولى وإيه اللي كان فيها

أول تعريف لليد كان كده:

typedef enum { LED_OFF, LED_ON } led_status_t;

typedef struct {
    uint8 port_name  : 4;
    uint8 pin        : 3;
    uint8 led_status : 1;
} led_t;

Std_ReturnType led_initialize(const led_t *led) {
    Std_ReturnType ret = E_OK;
    if (NULL == led) {
        ret = E_NOT_OK;
    } else {
        pin_config_t pin_obj = { .port = led->port_name, .pin = led->pin,
                                 .direction = GPIO_DIRECTION_OUTPUT,
                                 .logic = led->led_status };
        gpio_pin_initialize(&pin_obj);
    }
    return ret;
}

الكود ده بيشتغل. بس فيه تلات حاجات، واحدة منهم غلطة حقيقية.

1. نتيجة الدالة اتنسيت

السطر gpio_pin_initialize(&pin_obj); بيرجّع E_OK أو E_NOT_OK، والنتيجة رايحة في الهوا. يعني لو حد عمل ليد على PORTE رجل 6، الـ GPIO هيرفض، و led_initialize هترجع E_OK كأن كل حاجة تمام. الأبلكيشن هيصدّق، والليد عمرها ما هتنوّر، ومفيش أي حاجة تقولك ليه. الحل كلمة واحدة: ret = gpio_pin_initialize(&pin_obj);

خلي بالكلو قصدك تتجاهل النتيجة، اكتبها صريحة: (void)gpio_init();. كده اللي بيقرا الكود (وأدوات الفحص زي MISRA) يعرف إنك شايفها وقررت تسيبها، مش ناسيها. هتلاقي ده في application.c وفي المشاريع.

2. struct على الـ stack من غير لازمة

بنبني pin_config_t مؤقت، بس علشان نبعته لدالة هتفكّه تاني. الـ Direct API بيعمل نفس الشغل بنداء واحد:

ret = gpio_pin_setup((port_index_t)led->port, (pin_index_t)led->pin,
                     GPIO_DIRECTION_OUTPUT, GPIO_LOW);

الفرق مش كبير في الليد الواحدة: بايت على الـ stack وكام تعليمة. بس الشريحة دي فيها stack للـ hardware بعمق 31 نداء بس، والـ RAM 4KB، ودالة زي دي ممكن تتنادى جوّه لوب سريع. العادة الصح من الأول أرخص.

3. اسم led_status بيلخبط

هو كان بيتستخدم كقيمة أولية للرجل (LOW أو HIGH)، مش حالة الليد. ولو الليد Sink، الـ LOW هو اللي بينوّر، فـ LED_OFF اللي قيمتها 0 هتنوّر الليد. يعني الاسم بيكدب في نص الحالات.

النسخة اللي في الريبو دلوقتي

typedef enum { LED_ACTIVE_HIGH_MODE = 0, LED_ACTIVE_LOW_MODE = 1 } led_active_t;
typedef enum { LED_OFF = 0, LED_ON = 1 } led_state_t;

typedef struct {
    uint8 port     : 3;
    uint8 pin      : 3;
    uint8 active   : 1;    /* Active High ولا Low */
    uint8 reserved : 1;    /* محجوز لبعدين */
} led_config_t;

اتغيّر إيه؟ port بقى 3 بت زي pin_config_t بالظبط، فالتحويل بينهم مبيقطعش حاجة. والحالة الأولية اتشالت خالص من الـ struct، لأن الليد بتبدأ مطفية دايمًا. ومكانها جه active: الليد Source ولا Sink. والبت الأخير reserved واخد مكانه علشان الـ struct يفضل بايت كامل، ولو احتجنا حاجة بعدين (زي "ليد بتومض لوحدها") في مكان ليها.

المنطق المقلوب في ماكرو واحد

#define LED_ON_LOGIC(led)   (((led)->active == LED_ACTIVE_HIGH_MODE) ? GPIO_HIGH : GPIO_LOW)
#define LED_OFF_LOGIC(led)  (((led)->active == LED_ACTIVE_HIGH_MODE) ? GPIO_LOW  : GPIO_HIGH)

كل حتة في الدرايفر بتسأل الماكرو ده بدل ما تفكر. led_on بتكتب LED_ON_LOGIC، وخلاص. والأبلكيشن عمره ما هيعرف الليد متوصلة إزاي.

الاتنين API، والاختيار وقت الكومبايل

#if GPIO_DIRECT_API_ENABLE == CONFIG_ENABLE
    #define LED_USE_GPIO_DIRECT_API
#endif

Std_ReturnType led_initialize(const led_config_t *led) {
    Std_ReturnType ret = E_OK;
    if (NULL == led) {
        ret = E_NOT_OK;
    } else {
#ifdef LED_USE_GPIO_DIRECT_API
        ret = gpio_pin_setup((port_index_t)led->port, (pin_index_t)led->pin,
                             GPIO_DIRECTION_OUTPUT, LED_OFF_LOGIC(led));
#else
        pin_config_t cfg = led_to_pin_config(led, LED_OFF_LOGIC(led));
        ret = gpio_pin_initialize(&cfg);
#endif
    }
    return ret;
}

الدرايفر بيفضّل الـ Direct لو مفتوح، ولو مقفول بيرجع للـ Struct. والاتنين بياخدوا الـ ret. ولو الاتنين مقفولين، الـ #error اللي في device_config.h بيوقف الكومبايل قبل ما توصل هنا.

وبص على LED_OFF_LOGIC(led) في الـ setup: ليد Sink هتتعمل وهي HIGH، يعني مطفية، ومش هتغمز لحظة التشغيل. ده نفس موضوع ترتيب LAT قبل TRIS، بس من فوق.

led_get_state بتقرا منين؟

من الـ LAT، بـ gpio_pin_read_latch. لو قريت PORT ممكن تلاقي الجهد واقع شوية بسبب الليد نفسها وتتقري غلط. وإحنا عايزين نعرف "أنا قلتلها تنوّر ولا لأ"، وده مكانه الـ Latch.

من الأبلكيشن

led_config_t led_red = { .port = PORTC_INDEX, .pin = GPIO_PIN0,
                         .active = LED_ACTIVE_HIGH_MODE, .reserved = 0 };

void main(void) {
    (void)gpio_init();
    led_initialize(&led_red);
    while (1) {
        led_on(&led_red);
        __delay_ms(500);
        led_off(&led_red);
        __delay_ms(500);
    }
}

مفيش ولا ريجستر. لو نقلت الليد لـ RD5 وقلبتها Sink، بتغيّر سطر التعريف بس.

جرّب تكمّل بنفسك

الدرايفر مفيهوش دالة led_set(led, state) بتاخد ON أو OFF كباراميتر. حاول تكتبها في 6 سطور: فحص الـ NULL، وفحص إن الـ state صح، وبعدين تنادي led_on أو led_off وترجّع نتيجتها. لو عملتها، هتلاقي الـ Morse ممكن يتكتب أقصر.

الدرايفر ده بيتكتب من الصفر خطوة خطوة في درايفر الـ LED من الصفر.

ليد Active Low على RD2. ندهت led_initialize. الرجل هتبقى إيه؟

LED_OFF_LOGIC لليد Active Low بيدّي GPIO_HIGH. الرجل HIGH والطرف التاني على VCC، يبقى مفيش تيار.

في النسخة القديمة، إيه أخطر مشكلة؟

فحص الـ NULL كان موجود، والـ struct كان بايت. بس الدالة كانت ترجع E_OK حتى لو الـ GPIO رفض.
