عندنا GPIO وليد. نعمل بيهم حاجة حقيقية من الصفر، ونمشي فيها بنفس الطريقة: نسخة ساذجة، نشوفها بتقع فين، نصلّح. المشروع: زرار واحد، وكل ضغطة تنقل ليد منوّرة لليد اللي بعدها على 4 ليدات، وضغطة طويلة (ثانية) ترجّع للأولى.

التوصيل: 4 ليدات على RD0..RD3 (Source، 330Ω)، وزرار بين RB0 والأرضي، والـ pull-ups الداخلية.

v1: اقرا واعمل

static const led_config_t leds[4] = {
    { .port = PORTD_INDEX, .pin = GPIO_PIN0, .active = LED_ACTIVE_HIGH_MODE },
    { .port = PORTD_INDEX, .pin = GPIO_PIN1, .active = LED_ACTIVE_HIGH_MODE },
    { .port = PORTD_INDEX, .pin = GPIO_PIN2, .active = LED_ACTIVE_HIGH_MODE },
    { .port = PORTD_INDEX, .pin = GPIO_PIN3, .active = LED_ACTIVE_HIGH_MODE },
};

void main(void) {
    uint8   cur = 0, i;
    logic_t b;

    (void)gpio_init();
    (void)gpio_portb_pullup_enable();
    (void)gpio_pin_setup(PORTB_INDEX, GPIO_PIN0, GPIO_DIRECTION_INPUT, GPIO_LOW);
    for (i = 0; i < 4; i++) { (void)led_initialize(&leds[i]); }
    (void)led_on(&leds[0]);

    while (1) {
        (void)gpio_pin_read(PORTB_INDEX, GPIO_PIN0, &b);
        if (GPIO_LOW == b) {
            (void)led_off(&leds[cur]);
            cur = (uint8)((cur + 1) % 4);
            (void)led_on(&leds[cur]);
        }
    }
}

حرّقه ودوس. الليد هتجري عشوائي، ومش هتعرف توقّفها على واحدة. السبب إن اللوب بيلف مئات الألوف من المرات في الثانية، وطول ما صباعك على الزرار (حوالي 100ms) الشرط صح، فالليد بتتنقل آلاف المرات. وحتى لو كان سريع جدًا، الـ bounce كان هيعمل نفس الحاجة.

بالمناسبة، ليه الـ array const؟ عشان يتحط في الـ Flash. والدوال بتاخد const led_config_t * فده ماشي من غير cast. ده سبب إضافي إن الـ API بيقبل const.

v3: Debounce بالعد

الهدف: منصدّقش التغيير غير لو ثبت 20ms.
#define TICK_MS         5
#define DEBOUNCE_TICKS  4      /* 4 × 5ms = 20ms */

typedef struct {
    logic_t stable;
    uint8   counter;
} debounce_t;

static boolean_t pressed_edge(debounce_t *d, logic_t raw) {
    boolean_t edge = FALSE;
    if (raw != d->stable) {
        if (++d->counter >= DEBOUNCE_TICKS) {
            d->stable  = raw;
            d->counter = 0;
            edge = (GPIO_LOW == raw) ? TRUE : FALSE;
        }
    } else {
        d->counter = 0;
    }
    return edge;
}

ده نفس المنطق اللي في مشروع العداد في الريبو. وفي اللوب نقرا كل 5ms بس:

    debounce_t btn = { GPIO_HIGH, 0 };
    while (1) {
        (void)gpio_pin_read(PORTB_INDEX, GPIO_PIN0, &b);
        if (pressed_edge(&btn, b)) {
            (void)led_off(&leds[cur]);
            cur = (uint8)((cur + 1) % 4);
            (void)led_on(&leds[cur]);
        }
        __delay_ms(TICK_MS);
    }

v4: الضغطة الطويلة

الهدف: لو الزرار فضل مضغوط ثانية، ارجع لليد الأولى.
+#define LONG_TICKS      (1000 / TICK_MS)   /* 200 */
+    uint16 held = 0;
     while (1) {
         (void)gpio_pin_read(PORTB_INDEX, GPIO_PIN0, &b);
         if (pressed_edge(&btn, b)) { ... }
+        if (GPIO_LOW == btn.stable) {
+            if (held < LONG_TICKS) {
+                if (++held == LONG_TICKS) {
+                    (void)led_off(&leds[cur]);
+                    cur = 0;
+                    (void)led_on(&leds[cur]);
+                }
+            }
+        } else {
+            held = 0;
+        }
         __delay_ms(TICK_MS);
     }

ليه uint16؟ 200 بتدخل في uint8، بس لو حد غيّر TICK_MS لـ 2 تبقى 500، وتعدّي 255 من غير ما حد ياخد باله. ليه held < LONG_TICKS قبل الزيادة؟ علشان العداد يقف عند 200 وميلفّش، وإلا لو فضلت ضاغط هيلف ويرجع للأولى تاني كل كام ثانية. وخد بالك إن الضغطة الطويلة بتبدأ بضغطة عادية، فالليد هتتنقل خطوة الأول وبعدين ترجع للأولى. لو مش عايز ده، الحل إنك تنفّذ الضغطة القصيرة لحظة الرفع بدل النزول. جرّبها كتمرين.

تمرين: الضغطة القصيرة عند الرفع

عدّل المشروع: الضغطة القصيرة (أقل من ثانية) تنقل الليد لما تشيل صباعك، والطويلة ترجّع للأولى من غير ما تنقل الأول.

اعرض الحل 
static boolean_t changed(debounce_t *d, logic_t raw) {   /* TRUE عند أي تغيير مستقر */
    boolean_t c = FALSE;
    if (raw != d->stable) {
        if (++d->counter >= DEBOUNCE_TICKS) { d->stable = raw; d->counter = 0; c = TRUE; }
    } else { d->counter = 0; }
    return c;
}

/* في اللوب */
if (changed(&btn, b) && (GPIO_HIGH == btn.stable)) {      /* لحظة الرفع */
    if (held < LONG_TICKS) { next_led(); }                 /* كانت قصيرة */
    held = 0;
}
if (GPIO_LOW == btn.stable && held < LONG_TICKS) {
    if (++held == LONG_TICKS) { go_first(); }
}

الفكرة إن القرار بيتاخد لما الضغطة تخلص وتعرف طولها. ده نفس اللي الموبايل بيعمله في زرار الباور.
