أكبر غلطة في كتابة الدرايفرات إنك تفتح الـ .c وتبدأ تكتب. الـ API (أسامي الدوال، والباراميترز، وبترجّع إيه) هو العقد بينك وبين كل اللي هيستخدم الدرايفر، وإنت منهم بعد شهر. لو العقد وحش، الكود الحلو جوّاه مش هينقذه. فالدرس ده كله في ملف الـ .h، ومفيش ولا سطر تنفيذ.

الخطوة 1: نكتب الاستخدام الأول

الهدف: نعرف عايزين نكتب إيه في الأبلكيشن، قبل ما نعرف الدرايفر هيعمله إزاي.

دي حيلة بسيطة جدًا وبتوفّر كتير: اكتب الكود اللي نفسك تكتبه، كأن الدرايفر موجود.

gpio_pin_setup(PORTC_INDEX, GPIO_PIN0, GPIO_DIRECTION_OUTPUT, GPIO_LOW);
gpio_pin_write(PORTC_INDEX, GPIO_PIN0, GPIO_HIGH);
gpio_pin_toggle(PORTC_INDEX, GPIO_PIN0);

logic_t btn;
if (E_OK == gpio_pin_read(PORTB_INDEX, GPIO_PIN0, &btn) && GPIO_LOW == btn) { ... }

من الأربع سطور دول طلع كل حاجة: محتاجين أنواع للبورت والرجل والاتجاه والقيمة. القراية لازم ترجّع حاجتين (نجحت؟ والقيمة)، فالقيمة هتطلع في مؤشر والنتيجة في الـ return. والأسامي كلها بتبدأ بـ gpio_ علشان C مفيهاش namespaces، والبادئة هي اللي بتمنع إن pin_write بتاعتك تتخانق مع واحدة في مكتبة تانية.

الخطوة 2: الأنواع

typedef enum { GPIO_LOW = 0, GPIO_HIGH = 1 } logic_t;
typedef enum { GPIO_DIRECTION_OUTPUT = 0, GPIO_DIRECTION_INPUT = 1 } direction_t;
typedef enum { GPIO_PIN0 = 0, GPIO_PIN1, GPIO_PIN2, GPIO_PIN3,
               GPIO_PIN4, GPIO_PIN5, GPIO_PIN6, GPIO_PIN7 } pin_index_t;
typedef enum { PORTA_INDEX = 0, PORTB_INDEX, PORTC_INDEX, PORTD_INDEX, PORTE_INDEX } port_index_t;

ليه enum مش #define؟ الـ debugger بيوريك اسم القيمة بدل الرقم. والأهم إن نوع الباراميتر بيوصف معناه: direction_t direction بتقولك إيه اللي المفروض يتبعت. الـ C مش هيمنعك تبعت 7 (الـ enums في C ضعيفة)، بس أدوات الفحص والمراجع هيشوفوها على طول.

ليه القيم دي بالظبط؟ OUTPUT = 0 لأن بت الـ TRIS بيبقى 0 للـ output. يعني القيمة ممكن تتكتب في الريجستر من غير if. والبورتات من 0 لـ 4 علشان تبقى index في جدول.

الخطوة 3: النتيجة

كل دالة بترجّع Std_ReturnType. حتى اللي "مستحيل تفشل" زي gpio_portb_pullup_enable. ليه؟ علشان الشكل يبقى واحد، ولو بكرة نقلناها لشريحة فيها شروط، الكود اللي بينادي مش هيتغير.

خلي بالكفيه مدرسة تانية بتقول الدرايفر يرجّع void ويعمل assert لو فيه غلط، علشان السرعة. ده منطقي في كود الإنتربت. إحنا اخترنا إن الفحص يفضل موجود والنتيجة ترجع، لأن 4 مقارنات على شريحة 4MHz حوالي 4 ميكروثانية، وده رخيص جدًا قصاد ساعة ديباج.

الخطوة 4: ليه اتنين API؟

الـ Direct اللي فوق حلو للاستخدام السريع. بس تخيل مشروع فيه 15 رجل لكل واحدة إعداد ثابت. هتكتب 15 سطر setup، وكل مرة تكتب فيها لرجل هتكتب البورت والرجل تاني. لو رجل اتنقلت، هتدوّر عليها في كل حتة. الحل: كل رجل توصفها مرة في struct.

typedef struct {
    uint8 port      : 3;
    uint8 pin       : 3;
    uint8 direction : 1;
    uint8 logic     : 1;
} pin_config_t;

Std_ReturnType gpio_pin_initialize(const pin_config_t *_pin_config);
Std_ReturnType gpio_pin_write_logic(const pin_config_t *_pin_config, logic_t logic);

ليه const pin_config_t *؟ مؤشر علشان بنبعت عنوان (بايتين) بدل ما ننسخ الـ struct. والـ const وعد إن الدالة مش هتغيّر فيه، وده بيخليك تعرّف الإعدادات نفسها const فتتحط في الـ Flash.

ليه bitfields؟ 3 بت للبورت (0–7، محتاجين 0–4) و 3 للرجل و 1 و 1. المجموع بايت. من غيرها 4 بايت. مع 15 رجل الفرق 45 بايت RAM، وده 1% من الـ RAM كلها. مش كتير لوحده، بس دي عادة.

الاسم مختلف (_logic في الآخر) علشان الاتنين يعيشوا مع بعض من غير تعارض.

الخطوة 5: مفاتيح وقت الكومبايل

الهدف: كل API يتفتح ويتقفل لوحده، ومفيش واحد يعتمد على التاني.
/* device_config.h */
#define GPIO_MODULE_ENABLE              CONFIG_ENABLE
#define GPIO_STRUCT_API_ENABLE          CONFIG_ENABLE
#define GPIO_DIRECT_API_ENABLE          CONFIG_ENABLE
#define GPIO_PORT_PIN_CONFIGURATIONS    CONFIG_ENABLE
#define GPIO_PORT_CONFIGURATIONS        CONFIG_ENABLE

/* hal_gpio.h */
#if GPIO_MODULE_ENABLE == CONFIG_ENABLE
    Std_ReturnType gpio_init(void);
  #if GPIO_DIRECT_API_ENABLE == CONFIG_ENABLE
    Std_ReturnType gpio_pin_setup(...);
    ...
  #endif
  #if GPIO_STRUCT_API_ENABLE == CONFIG_ENABLE
    #if GPIO_PORT_PIN_CONFIGURATIONS == CONFIG_ENABLE
      Std_ReturnType gpio_pin_initialize(...);
    #endif
  #endif
#endif

الجملة المهمة: "مفيش واحد يعتمد على التاني". كان سهل نكتب الـ Struct API إنه يفك الـ struct وينادي الـ Direct. بس ساعتها لو حد قفل الـ Direct علشان يوفّر Flash، الـ Struct هيقع. فالاتنين بيستخدموا helpers داخلية مشتركة (static) ومحدش فيهم بينادي التاني. هنشوف ده في الكود.

الخطوة 6: ماكروز الفحص في الـ .h

#define PORT_PIN_MAX_NUMBER         8
#define PORT_MAX_NUMBER             5
#define PORTE_PIN_MAX_NUMBER        3
#define IS_VALID_PORT(port)         (((uint8)(port)) < PORT_MAX_NUMBER)
#define IS_VALID_PIN(pin)           (((uint8)(pin)) < PORT_PIN_MAX_NUMBER)
#define IS_VALID_DIRECTION(dir)     (((uint8)(dir)) <= GPIO_DIRECTION_INPUT)
#define IS_VALID_LOGIC(logic)       (((uint8)(logic)) <= GPIO_HIGH)

ليه الـ cast لـ uint8؟ لأن لو حد بعت قيمة سالبة (enum في C ممكن يبقى signed)، المقارنة < 5 هتعدّي. بعد الـ cast، الـ −1 بقت 255 والفحص يمسكها. ومفيش فحص >= 0 خالص، الـ unsigned بيغني عنه.

ليه الـ Struct API مبينادي الـ Direct API جوّاه؟

الاستقلال بين الاتنين هو اللي بيخلي المفاتيح حقيقية، مش ديكور.

تمرين: صمم API لزرار

قبل ما تبص في أي حاجة: اكتب الـ .h بتاع درايفر زرار (ECU). محتاج تعرف: الزرار على أنهي رجل، متوصل pull-up ولا pull-down، وتقرا حالته (مضغوط ولا لأ). اكتب الأنواع والدوال بس.

اعرض حل مقترح 
typedef enum { BUTTON_RELEASED = 0, BUTTON_PRESSED = 1 } button_state_t;
typedef enum { BUTTON_ACTIVE_LOW = 0, BUTTON_ACTIVE_HIGH = 1 } button_active_t;

typedef struct {
    uint8 port   : 3;
    uint8 pin    : 3;
    uint8 active : 1;
    uint8 reserved : 1;
} button_config_t;

Std_ReturnType button_initialize(const button_config_t *btn);
Std_ReturnType button_read_state(const button_config_t *btn, button_state_t *state);

لاحظ إن الأبلكيشن مش هيشوف LOW و HIGH خالص، هيشوف PRESSED و RELEASED. ده نفس اللي عملناه في الليد. والـ debounce؟ محتاج حالة بتتحفظ بين النداءات، فهيحتاج struct تاني مش const. فكّر فيها، هنرجعلها في المشروع.
