كتبت الكود صح، والبرنامج اتحرق، والليد مش بتومض. أول حاجة تبص عليها مش الكود، الفيوزات. الـ Configuration Bits هي الإعدادات اللي الشريحة بتقراها قبل ما تنفذ أول تعليمة، وأهمها: الساعة جاية منين.

الخطوة 1: نطلّع الإعدادات من MPLAB

الهدف: ملف فيه كل الفيوزات بقيم إحنا اخترناها، مش الافتراضي.

Window ← Target Memory Views ← Configuration Bits. هيفتحلك جدول بكل إعداد وقيمته. غيّر اللي محتاجه، ودوس Generate Source Code to Output، وانسخ الناتج في ملف جديد Config/config_bits.c.

#pragma config OSC = HS
#pragma config WDT = OFF
#pragma config LVP = OFF
#pragma config PBADEN = OFF
#pragma config MCLRE = ON
#pragma config BOREN = OFF
#pragma config XINST = OFF

ليه في ملف لوحده؟ لأن الـ XC8 بيقبل #pragma config في ملف C واحد بس في البروجكت. لو حطيتهم في application.c، وكل مشروع من المشاريع الأربعة عنده main.c خاص بيه، هتنسخهم أربع مرات، ولو غيّرت حاجة هتنسى نسخة. ملف واحد بيتضاف لأي بروجكت مرة واحدة.

ليه كل واحد فيهم كده

الخطوة 2: الإعدادات تيجي من مكان واحد

الهدف: نغيّر الكريستالة في سطر واحد، والفيوز يتغير لوحده.

النسخة الأولى فيها مشكلة خفية: _XTAL_FREQ في ملف، و OSC = HS في ملف تاني. حد هيغيّر الكريستالة لـ 8MHz بالـ PLL ويغيّر واحد وينسى التاني. نعمل ملف Config/device_config.h وكل القرارات تبقى فيه:

#define _XTAL_FREQ          4000000UL

#define OSC_MODE_XT         1
#define OSC_MODE_HS         2
#define OSC_MODE_HSPLL      3
#define OSC_MODE_INTIO67    4
#define OSC_MODE            OSC_MODE_HS

وفي config_bits.c:

+#include "device_config.h"
-#pragma config OSC = HS
+#if   OSC_MODE == OSC_MODE_XT
+#pragma config OSC = XT
+#elif OSC_MODE == OSC_MODE_HS
+#pragma config OSC = HS
+#elif OSC_MODE == OSC_MODE_HSPLL
+#pragma config OSC = HSPLL
+#elif OSC_MODE == OSC_MODE_INTIO67
+#pragma config OSC = INTIO67
+#endif

الخطوة 3: الملف يرفض الغلط

الهدف: لو الإعدادات مش منطقية مع بعض، الـ build يقف ويقول ليه.

الـ HSPLL بيضرب الكريستالة في 4. يعني كريستالة 10MHz تدّيك 40MHz. لو حد كتب HSPLL و _XTAL_FREQ = 4000000، الـ delays هتبقى أطول أربع مرات. نخلي الكومبايلر يمسكها:

+#if !defined(OSC_MODE)
+    #error "OSC_MODE is not defined in device_config.h"
+#elif OSC_MODE == OSC_MODE_XT
+    #if (_XTAL_FREQ > 4000000UL)
+        #error "XT mode supports crystals up to 4 MHz - use OSC_MODE_HS"
+    #endif
+#elif OSC_MODE == OSC_MODE_HS
+    #if (_XTAL_FREQ < 4000000UL) || (_XTAL_FREQ > 25000000UL)
+        #error "HS mode needs a 4 - 25 MHz crystal (_XTAL_FREQ)"
+    #endif
+#elif OSC_MODE == OSC_MODE_HSPLL
+    #if (_XTAL_FREQ < 16000000UL) || (_XTAL_FREQ > 40000000UL)
+        #error "HSPLL: _XTAL_FREQ = 4 x crystal (4 - 10 MHz) -> must be 16 - 40 MHz"
+    #endif
+#endif

ده نفس اللي في الريبو. والـ INTIO67 ليه فحص لوحده بيتأكد إن السرعة واحدة من اللي الأوسيليتور الداخلي يقدر يطلّعها (31k، 125k، … 8M).

خلي بالكالـ #error بيشتغل وقت الكومبايل، مش وقت التشغيل. يعني مبياخدش ولا بايت من الـ Flash، ومبيبطّأش حاجة. أي حاجة تقدر تمسكها بـ #error، امسكها بيه بدل ما تكتشفها على البورد.
غلطات مشهورة

نسيت تضيف config_bits.c للبروجكت: الشريحة هتشتغل بالفيوزات الافتراضية (OSC = RC غالبًا)، يعني الساعة جاية من مكان مش متوصل، ومش هتشتغل أو هتشتغل بسرعة عشوائية.

ضفته مرتين (مرة من الريبو ومرة نسخة): خطأ بإن الإعدادات متعرّفة أكتر من مرة.

الـ MCLRE ON ومفيش مقاومة على MCLR: الشريحة واقفة في ريست دايمًا. على البورد ده شكله "مفيش أي حاجة بتحصل".

كريستالة 8MHz، و OSC_MODE = HSPLL. تكتب _XTAL_FREQ كام؟

الـ PLL بيضرب في 4، والـ delays محتاجة السرعة اللي المعالج شغال بيها فعلًا: 32MHz.

تمرين: نشغّل من غير كريستالة

معاكش كريستالة. عايز تشتغل على الأوسيليتور الداخلي 8MHz. غيّر إيه في device_config.h؟ وإيه اللي لازم يتعمل في الكود كمان؟

اعرض الحل 
#define _XTAL_FREQ  8000000UL
#define OSC_MODE    OSC_MODE_INTIO67

/* وفي أول main: اختار 8MHz من الأوسيليتور الداخلي */
OSCCONbits.IRCF = 0x7;    /* 111 = 8 MHz */

الفيوز بيقول "الساعة جاية من جوّه"، بس سرعتها نفسها بتتحدد من ريجستر OSCCON وقت التشغيل، والافتراضي بعد الريست 1MHz. لو نسيت السطر ده، كل الـ delays هتبقى أبطأ 8 مرات. وبالمرة RA6 و RA7 بقوا فاضيين ليك تستخدمهم I/O — والدرايفر ساعتها بس بيوافق عليهم (IS_VALID_PORT_PIN من غير INTIO67 بترجع E_NOT_OK على الرجول دي).
