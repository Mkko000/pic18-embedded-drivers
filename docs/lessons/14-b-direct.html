v1: switch على البورت

الهدف: gpio_pin_write تشتغل. وخلاص.
Std_ReturnType gpio_pin_write(port_index_t port, pin_index_t pin, logic_t logic) {
    switch (port) {
        case PORTA_INDEX: if (logic) PORTA |= (1 << pin); else PORTA &= ~(1 << pin); break;
        case PORTB_INDEX: if (logic) PORTB |= (1 << pin); else PORTB &= ~(1 << pin); break;
        case PORTC_INDEX: if (logic) PORTC |= (1 << pin); else PORTC &= ~(1 << pin); break;
        case PORTD_INDEX: if (logic) PORTD |= (1 << pin); else PORTD &= ~(1 << pin); break;
        case PORTE_INDEX: if (logic) PORTE |= (1 << pin); else PORTE &= ~(1 << pin); break;
    }
    return E_OK;
}

جرّبها على ليد: شغالة. فين المشكلة؟ تلاتة.

1. بنكتب في PORT. فاكر حكاية read-modify-write من أول درس؟ PORTC |= ... معناها: اقرا PORTC (الجهد الحقيقي على الرجول)، غيّر بت، اكتب الكل. لو فيه رجل تانية عليها حمل تقيل أو مكثف وجهدها لسه ماطلعش، هتتقري 0 وتتكتب 0.

جرّبها كده: RC0 عليها ليد ومكثف 10µF بالتوازي (بيمثّل حمل بطيء). اكتب RC0 = 1 وعلى طول في السطر اللي بعده RC1 = 1. في v1 هتلاقي RC0 ساعات بتطفي لوحدها. نفس التجربة بـ LATC، عمرها ما هتحصل.

2. الكود متكرر 5 مرات. ده مش بس شكل. كل دالة هتكتبها (read، toggle، direction) هيبقى فيها نفس الـ switch. 12 دالة × 5 حالات = 60 مكان ممكن تغلط في واحد فيهم.

3. مفيش فحص. لو pin = 9، 1 << 9 بـ 512، والـ cast لبايت بيدّي 0، فمفيش حاجة هتحصل والدالة هتقول E_OK.

كلمة volatile، بالتجربة

شيل volatile من جدول PORT واكتب لوب مستني زرار:

logic_t b = GPIO_HIGH;
do { gpio_pin_read(PORTB_INDEX, GPIO_PIN0, &b); } while (GPIO_HIGH == b);

الـ XC8 المجاني غالبًا مش هيلاحظ، لأن الـ optimization ضعيف. بس على مستوى optimization أعلى أو كومبايلر تاني، الكومبايلر ممكن يستنتج إن محدش بيكتب في PORTB، فيقراه مرة ويحط القيمة في ريجستر ويلف عليها للأبد. الـ volatile بتمنعه. القاعدة: أي مؤشر لريجستر هاردوير يبقى volatile. مفيش استثناء.

والـ const في الآخر

volatile uint8 * const: الـ const بعد النجمة يعني المؤشر نفسه ثابت. الجدول مش هيتغير، فالكومبايلر يحطه في الـ Flash وميضيّعش RAM.

ليه متغير ret و return واحد في الآخر بدل ما نعمل return E_NOT_OK على طول؟ دي قاعدة من MISRA (single exit point). ممكن تختلف معاها، وفيه ناس شاطرة بتختلف معاها. بس ليها فايدة عملية: لو بكرة عايز تضيف حاجة تتعمل في الآخر دايمًا (زي إنك تعدّ الأخطاء)، فيه مكان واحد.

وليه IS_VALID_PORT_PIN مش IS_VALID_PORT && IS_VALID_PIN؟ لأن PORTE رجله 5 بتعدّي الفحص التاني، والريجستر LATE بت 5 موجود في الذاكرة بس مالوش رجل. ونفس الحكاية مع RA6 و RA7 لو إنت على كريستالة: البتات في LATA موجودة، بس الرجول مش I/O (هما OSC2/OSC1). مش هيبوظ حاجة، بس الدالة هتقول E_OK على حاجة مستحيلة.

خلي بالكالغلط ده بيعدّي الكومبايل عادي: gpio_pin_write(PORTA_INDEX, GPIO_PIN7, ...) شكله سليم. اللي بيمسكك وقت التشغيل هو IS_VALID_PORTA_PIN: مع XT/HS/HSPLL بيرفض أي pin ≥ 6 على PORTA وبيرجع E_NOT_OK، ومع OSC_MODE_INTIO67 بيوافق على الكل. ده نفس شرط ماسك PORTA 0x3F / 0xFF في دوال البورت.

v4: gpio_pin_setup والترتيب

الهدف: نداء واحد يجهّز الرجل، من غير glitch.

أول نسخة طبيعية:

if (GPIO_HIGH == logic) SET_BIT(*(lat_registers[port]), pin);
else                    CLEAR_BIT(*(lat_registers[port]), pin);

v5: القراية، ومؤشر NULL

Std_ReturnType gpio_pin_read(port_index_t port, pin_index_t pin, logic_t *logic) {
    Std_ReturnType ret = E_OK;
    if (NULL == logic) {
        ret = E_NOT_OK;
    }
    else if (!IS_VALID_PORT_PIN(port, pin)) {
        ret = E_NOT_OK;
    }
    else {
        *logic = (READ_BIT(*(port_registers[port]), pin) == 1) ? GPIO_HIGH : GPIO_LOW;

فحص NULL الأول. لو حد بعت NULL، الكتابة في *logic على الـ PIC مش هتعمل crash زي الكمبيوتر، هتكتب في عنوان 0 في الـ RAM، وده متغير حقيقي عندك. الكمبيوتر بيرحمك بالـ crash، الميكرو مبيرحمش.

ليه ? GPIO_HIGH : GPIO_LOW مش *logic = READ_BIT(...) على طول؟ الاتنين بيدّوا نفس الرقم. بس الأولى بتحوّل لـ enum بشكل صريح، ولو في يوم GPIO_HIGH بقى قيمته حاجة تانية (مستحيل هنا، بس الفكرة) الكود يفضل صح.

و gpio_pin_read_latch نفس الدالة بالظبط بس من lat_registers. ليه نحتاجها؟ علشان تعرف "أنا قلت إيه" من غير ما تحفظه في متغير. درايفر الليد هيستخدمها في led_get_state.

v6: التوجل

TOGGLE_BIT(*(lat_registers[port]), pin);

XOR على الـ LAT. لو عملته على PORT هترجع لمشكلة الـ read-modify-write، وأسوأ: لو الرجل Input، PORT بيرجع الجهد الخارجي، فالتوجل هيكتب عكس الجهد الخارجي في الـ Latch بدل عكس اللي إنت كاتبه.

نختبر كل خطوة

في الـ Simulator: حط breakpoint بعد كل نداء، وبص على TRISC و LATC في نافذة SFRs. اكتب اختبار صغير في main:

Std_ReturnType r;
r = gpio_pin_setup(PORTC_INDEX, GPIO_PIN0, GPIO_DIRECTION_OUTPUT, GPIO_HIGH);  /* E_OK, LATC.0=1, TRISC.0=0 */
r = gpio_pin_write(PORTE_INDEX, GPIO_PIN5, GPIO_HIGH);                         /* E_NOT_OK، LATE مايتغيرش */
r = gpio_pin_write(PORTA_INDEX, GPIO_PIN6, GPIO_HIGH);                         /* E_NOT_OK لو OSC مش INTIO67 */
r = gpio_pin_write(7, GPIO_PIN0, GPIO_HIGH);                                   /* E_NOT_OK */
r = gpio_pin_read(PORTB_INDEX, GPIO_PIN0, NULL);                               /* E_NOT_OK */
r = gpio_pin_toggle(PORTC_INDEX, GPIO_PIN0);                                   /* LATC.0 = 0 */

كل سطر جنبه اللي المفروض يحصل. لو حاجة مطابقتش، الغلط في الدالة دي بالظبط. ده unit test فقير، بس بيمسك 90% من الغلطات قبل ما توصل لبورد.

غلطات مشهورة

if (logic = GPIO_HIGH) بعلامة = واحدة: بيكتب مش بيقارن، والشرط دايمًا صح. علشان كده الكود كاتب الثابت على الشمال GPIO_HIGH == logic؛ لو نسيت = هيطلع خطأ كومبايل.

نسيت الـ *: SET_BIT(lat_registers[port], pin) بيعدّل المؤشر نفسه مش الريجستر، والكومبايلر هيشتكي لأن الجدول const. وده بالظبط ليه الـ const مفيد.

استخدمت pin كماسك بدل رقم: gpio_pin_write(PORTC_INDEX, GPIO_PIN3_MASK, ...). الماسك قيمته 8، والفحص هيرفضه. الماسكات للدوال بتاعة البورت بس.

ليه التوجل على LAT مش PORT؟

الكتابة في PORT مسموحة وبتروح للـ LAT، بس القراية جزء من الـ XOR، وهي اللي بتجيب الجهد الحقيقي.

إيه أخطر حاجة في v2 من غير فحص؟
