لو وصلت هنا وعملت التمارين اللي فاتت، إنت كتبت درايفر GPIO ودرايفر ليد ومشروع بيهم. التمارين دي أكبر شوية، وكل واحد بيجمع أكتر من درس. ابدأ بالمستوى اللي مريحك، ومتفتحش الحل غير بعد ما يبقى عندك كود بيتبني.

المستوى الأول

1. Knight Rider بالـ Struct API بس

اقفل GPIO_DIRECT_API_ENABLE. اعمل تأثير Knight Rider على PORTD (ليد واحدة رايحة جاية) باستخدام دوال البورت بتاعة الـ Struct API.

اعرض الحل 
void main(void) {
    uint8 pos = 0;
    sint8 dir = 1;
    (void)gpio_init();
    (void)gpio_port_initialize(PORTD_INDEX, GPIO_PORT_ALL_OUTPUT, GPIO_PORT_ALL_LOW);
    while (1) {
        (void)gpio_port_write_logic(PORTD_INDEX, (uint8)(1U << pos));
        __delay_ms(80);
        if ((7 == pos && dir > 0) || (0 == pos && dir < 0)) { dir = (sint8)-dir; }
        pos = (uint8)(pos + dir);
    }
}

لو اتبنى من غير أخطاء، ده إثبات إن المفاتيح مستقلة فعلًا.

2. قراية البورت بالحالة

4 زراير على RB4..RB7 (pull-ups). اعرض على RD4..RD7 مين مضغوط، بحيث الليد تنوّر طول ما زرارها مضغوط. سطرين في اللوب.

اعرض الحل 
uint8 v;
(void)gpio_port_read(PORTB_INDEX, &v);
(void)gpio_port_write(PORTD_INDEX, (uint8)(~v & 0xF0));

الـ ~ لأن الضغطة LOW، والـ & 0xF0 بيسيب بتات الزراير بس. خد بالك إن gpio_port_write بتكتب البايت كله، فالنص التحتاني من PORTD هيتصفّر كل لفة. لو عليه حاجة تانية، استخدم gpio_port_write_bits بتاعة تمرين درس البورت.

المستوى التاني

3. درايفر زرار كامل (ECU)

اكتب ECU_Layer/Button/ecu_button.h/.c: config فيه port و pin و active، و state فيه الـ debounce. دوال: button_initialize، و button_update (تتنادى كل 5ms)، و button_was_pressed (ترجع TRUE مرة واحدة لكل ضغطة). ضيف مفتاح BUTTON_MODULE_ENABLE (موجود فعلًا في device_config.h ومقفول، وفحص الـ #error بتاعه موجود).

اعرض حل مقترح 
typedef struct {
    uint8 port : 3;  uint8 pin : 3;  uint8 active : 1;  uint8 reserved : 1;
} button_config_t;

typedef struct {
    const button_config_t *cfg;
    logic_t stable;
    uint8   counter;
    uint8   pressed_flag;
} button_t;

Std_ReturnType button_initialize(button_t *btn, const button_config_t *cfg) {
    Std_ReturnType ret = E_OK;
    if ((NULL == btn) || (NULL == cfg)) {
        ret = E_NOT_OK;
    } else {
        btn->cfg = cfg;
        btn->stable = (BUTTON_ACTIVE_LOW == cfg->active) ? GPIO_HIGH : GPIO_LOW;
        btn->counter = 0;
        btn->pressed_flag = 0;
        ret = gpio_pin_setup((port_index_t)cfg->port, (pin_index_t)cfg->pin,
                             GPIO_DIRECTION_INPUT, GPIO_LOW);
    }
    return ret;
}

Std_ReturnType button_update(button_t *btn) {
    Std_ReturnType ret = E_OK;
    logic_t raw = GPIO_LOW;
    if (NULL == btn) { return E_NOT_OK; }
    ret = gpio_pin_read((port_index_t)btn->cfg->port, (pin_index_t)btn->cfg->pin, &raw);
    if (E_OK == ret) {
        if (raw != btn->stable) {
            if (++btn->counter >= 4) {
                btn->stable = raw; btn->counter = 0;
                if (raw == ((BUTTON_ACTIVE_LOW == btn->cfg->active) ? GPIO_LOW : GPIO_HIGH)) {
                    btn->pressed_flag = 1;
                }
            }
        } else { btn->counter = 0; }
    }
    return ret;
}

boolean_t button_was_pressed(button_t *btn) {
    boolean_t p = FALSE;
    if ((NULL != btn) && btn->pressed_flag) { btn->pressed_flag = 0; p = TRUE; }
    return p;
}

لاحظ الفصل: config ثابت (const، في الـ Flash)، و state بيتغير (في الـ RAM). وفي button_update فيه return بدري، ده كسر لقاعدة الـ single exit؛ صلّحه كجزء من التمرين.

4. 7-Segment

شاشة 7-segment Common Cathode على PORTD (a..g على RD0..RD6). اكتب جدول أرقام من 0 لـ 9، ودالة seg_show(uint8 digit). ولو الشاشة Common Anode؟ غيّر سطر واحد.

اعرض الحل 
static const uint8 seg_table[10] = {
    0x3F, 0x06, 0x5B, 0x4F, 0x66, 0x6D, 0x7D, 0x07, 0x7F, 0x6F
};
#define SEG_COMMON_ANODE  0

Std_ReturnType seg_show(uint8 digit) {
    Std_ReturnType ret = E_NOT_OK;
    if (digit < 10) {
        uint8 v = seg_table[digit];
#if SEG_COMMON_ANODE
        v = (uint8)~v;
#endif
        ret = gpio_port_write(PORTD_INDEX, (uint8)(v & 0x7F));
    }
    return ret;
}

نفس فكرة Active Low في الليد: المنطق المقلوب في مكان واحد. وحساب التيار: الرقم 8 فيه 7 أجزاء، لو كل واحد 10mA يبقوا 70mA من بورت واحد. تمام، بس لو 4 خانات بالـ multiplexing هتحتاج ترانزستورات على الـ common.
