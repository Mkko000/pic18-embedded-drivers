الـ GPIO خلص. دلوقتي أول درايفر في طبقة ECU. هتلاحظ إن الدرايفر ده تقريبًا مفيهوش "شغل"، كله نداءات للـ GPIO. ده بالظبط المقصود: كل الصعب اتعمل تحت، وهنا بنضيف المعنى.

v1: الأبسط

typedef struct {
    uint8 port_name  : 4;
    uint8 pin        : 3;
    uint8 led_status : 1;
} led_t;

Std_ReturnType led_turn_on(const led_t *led) {
    return gpio_pin_write(led->port_name, led->pin, GPIO_HIGH);
}

دي كانت أول نسخة في المشروع فعلًا (شفناها في درس "درايفر الـ LED"). نشوف بتقع فين:

1. مفيش فحص NULL. 2. port_name 4 بت والـ GPIO عنده 3: مش هيبوظ حاجة النهارده، بس الأنواع مش متطابقة. 3. GPIO_HIGH ثابتة: ليد Sink هتطفي لما تقولها نوّري. 4. led_status اسمه حالة، بس محدش بيحدّثه. لو حد قراه هيلاقي قيمة كدابة.

v3: ننادي أنهي API؟

الهدف: الليد تشتغل سواء الـ Direct مفتوح أو الـ Struct بس.
#if GPIO_DIRECT_API_ENABLE == CONFIG_ENABLE

#ifndef LED_USE_GPIO_DIRECT_API
static inline pin_config_t led_to_pin_config(const led_config_t *led, logic_t logic) {
    pin_config_t cfg;
    cfg.port      = led->port;
    cfg.pin       = led->pin;
    cfg.direction = GPIO_DIRECTION_OUTPUT;
    cfg.logic     = (uint8)logic;
    return cfg;
}
#endif

static Std_ReturnType led_write(const led_config_t *led, logic_t logic) {
#ifdef LED_USE_GPIO_DIRECT_API
    return gpio_pin_write((port_index_t)led->port, (pin_index_t)led->pin, logic);
#else
    pin_config_t cfg = led_to_pin_config(led, logic);
    return gpio_pin_write_logic(&cfg, logic);
#endif
}

القرار بيتاخد مرة واحدة فوق في ماكرو، وكل الدوال بتسأله. والتحويل لـ struct في helper واحد بدل ما يتكتب في كل دالة. و led_write نفسها static، يعني داخلية: on و off هيستخدموها.

ليه الـ Direct أولى لو الاتنين مفتوحين؟ لأن الليد عندها struct خاص بيها أصلًا، فتحويله لـ struct تاني علشان يتفك تاني تحت ده شغل على الفاضي.

v4: الدوال كلها

Std_ReturnType led_initialize(const led_config_t *led) {
    Std_ReturnType ret = E_OK;
    if (NULL == led) {
        ret = E_NOT_OK;
    }
    else {
#ifdef LED_USE_GPIO_DIRECT_API
        ret = gpio_pin_setup((port_index_t)led->port, (pin_index_t)led->pin,
                             GPIO_DIRECTION_OUTPUT, LED_OFF_LOGIC(led));
#else
        pin_config_t cfg = led_to_pin_config(led, LED_OFF_LOGIC(led));
        ret = gpio_pin_initialize(&cfg);
#endif
    }
    return ret;
}

Std_ReturnType led_on(const led_config_t *led) {
    return (NULL == led) ? E_NOT_OK : led_write(led, LED_ON_LOGIC(led));
}
Std_ReturnType led_off(const led_config_t *led) {
    return (NULL == led) ? E_NOT_OK : led_write(led, LED_OFF_LOGIC(led));
}

الـ ret = قدام نداء الـ GPIO هو بالظبط الغلطة اللي اتصلحت من النسخة الأولى. لو الـ GPIO رفض الرجل، الأبلكيشن هيعرف.

و led_get_state:

اللاتش مش PORT، والمقارنة بـ LED_ON_LOGIC مش بـ GPIO_HIGH، فليد Sink هتتقري صح. ومش بنكتب في *state لو القراية فشلت، علشان اللي بينادي ميلاقيش قيمة مش حقيقية.

اكتب led_blink_n(const led_config_t *led, uint8 times) تومض الليد عدد مرات، 100ms منوّرة و 100ms مطفية، وترجع الليد لحالتها الأصلية في الآخر.

اعرض الحل 
Std_ReturnType led_blink_n(const led_config_t *led, uint8 times) {
    Std_ReturnType ret = E_OK;
    led_state_t    was = LED_OFF;
    uint8          i;

    ret = led_get_state(led, &was);     /* بيفحص NULL كمان */
    for (i = 0; (E_OK == ret) && (i < times); i++) {
        ret = led_on(led);
        __delay_ms(100);
        if (E_OK == ret) { ret = led_off(led); }
        __delay_ms(100);
    }
    if ((E_OK == ret) && (LED_ON == was)) {
        ret = led_on(led);
    }
    return ret;
}

خد بالك: دالة فيها delay بتوقف كل حاجة. في ECU حقيقي، حاجة زي دي بتتعمل بتايمر. بس كتمرين على النتايج والحالة، دي كويسة. ومحتاجة _XTAL_FREQ، فلازم الملف يعمل include لـ device_config.h.
